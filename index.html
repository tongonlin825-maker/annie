function shuffle(arr) {

  const a = arr.slice();

  for (let i = a.length - 1; i > 0; i--) {

    const j = Math.floor(Math.random() * (i + 1));

    [a[i], a[j]] = [a[j], a[i]];

  }

  return a;

}

}

function playSound(ok) {

  try {

    if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

      o.frequency.value = f;

      g.gain.setValueAtTime(0.0001, now + i * 0.1);

      o.connect(g).connect(audioCtx.destination);

      o.start(now + i * 0.1);

      o.stop(now + i * 0.1 + 0.28);

    });

  } catch (e) {}

}

function saveRank(obj) {

function getBestStars(levelIndex) {

  const r = getRank();

  const arr = r[levelIndex] || [];

  if (!arr.length) return 0;

  return Math.max(...arr.map(x => x.stars));

}

function getBestTime(levelIndex) {

  const r = getRank();

  const arr = r[levelIndex] || [];

  if (!arr.length) return null;

  const best = arr.slice().sort((a, b) => a.time - b.time)[0];

  return best.time;

}

function addScore(levelIndex, rec) {

  const r = getRank();

  if (!r[levelIndex]) r[levelIndex] = [];

  r[levelIndex].push(rec);

  r[levelIndex].sort((a, b) => {

    if (b.stars !== a.stars) return b.stars - a.stars;

    return a.time - b.time;

  });

  r[levelIndex] = r[levelIndex].slice(0, 10);

  saveRank(r);

}

    state.currentSequence = lv.sequence.slice();

  } else if (lv.type === "listen") {

    state.currentQuestions = lv.questions.slice();

  }


  // 主題色

  document.documentElement.style.setProperty("--accent", lv.accent);

  document.documentElement.style.setProperty("--accent-light", lv.accentLight);


  $("gameTitle").textContent = lv.emoji + " " + lv.name + (state.mode === "retry" ? " · 錯題重練" : "");

  $("gameSub").textContent = lv.subtitle;

  $("instruction").textContent = "👉 " + lv.instruction;


  updateProgress();


  // 渲染棋盤

  const board = $("board");

  board.innerHTML = "";


  if (lv.type === "match") renderMatch(board);

  else if (lv.type === "order") renderOrder(board);

  else if (lv.type === "listen") renderListen(board);


  // 切換畫面

  $("home").classList.remove("active");

  $("rank").classList.remove("active");

  $("game").classList.add("active");

  window.scrollTo(0, 0);


  // 計時

  startTimer();

}


/* =========================================================

   計時器

   ========================================================= */

function startTimer() {

  stopTimer();

  state.startTime = Date.now();

  state.elapsed = 0;

  $("timer").textContent = "00:00";

  state.timerId = setInterval(() => {

    state.elapsed = Math.floor((Date.now() - state.startTime) / 1000);

    $("timer").textContent = fmtTime(state.elapsed);

  }, 250);

}

function stopTimer() {

  if (state.timerId) clearInterval(state.timerId);

  state.timerId = null;

  if (state.startTime) {

    state.elapsed = Math.floor((Date.now() - state.startTime) / 1000);

  }

}

   渲染：配對

   ========================================================= */

function renderMatch(board) {

  board.className = "board";

  const leftCol = document.createElement("div");

  leftCol.className = "column";

  leftCol.id = "colLeft";

  const rightCol = document.createElement("div");

  rightCol.className = "column";

  rightCol.id = "colRight";

  board.appendChild(leftCol);

  board.appendChild(rightCol);


  const pairs = state.currentPairs;

  const lefts = shuffle(pairs.map((p, i) => ({ text: p.left, id: i, small: p.small })));

  const rights = shuffle(pairs.map((p, i) => ({ text: p.right, id: i, small: p.small })));


  lefts.forEach(item => leftCol.appendChild(makeCard(item, "left")));

  rights.forEach(item => rightCol.appendChild(makeCard(item, "right")));


  board.onclick = (e) => {

    const card = e.target.closest(".card");

    if (!card) return;

    if (state.locked) return;

    if (card.classList.contains("matched")) return;

    onMatchCardClick(card);

  };

}


function makeCard(item, side) {

  const btn = document.createElement("button");

  btn.className = "card" + (side === "right" ? " card-jyut" : "") + (item.small ? " card-small" : "");

  btn.dataset.side = side;

  btn.dataset.id = item.id;

  btn.innerHTML = `<span class="card-text">${item.text}</span>`;

  return btn;

}


function onMatchCardClick(card) {

  const side = card.dataset.side;


  if (side === "left") {

    if (state.selectedLeft === card) {

      card.classList.remove("selected");

      state.selectedLeft = null;

      return;

    }

    if (state.selectedLeft) state.selectedLeft.classList.remove("selected");

    state.selectedLeft = card;

    card.classList.add("selected");

  } else {

    if (state.selectedRight === card) {

      card.classList.remove("selected");

      state.selectedRight = null;

      return;

    }

    if (state.selectedRight) state.selectedRight.classList.remove("selected");

    state.selectedRight = card;

    card.classList.add("selected");

  }


  if (state.selectedLeft && state.selectedRight) checkMatch();

}


function checkMatch() {

  const l = state.selectedLeft;

  const r = state.selectedRight;

  const pairs = state.currentPairs;

  const idL = +l.dataset.id;

  const idR = +r.dataset.id;

  const pair = pairs[idL];


  if (idL === idR) {

    l.classList.remove("selected");

    r.classList.remove("selected");

    l.classList.add("matched");

    r.classList.add("matched");

    state.selectedLeft = null;

    state.selectedRight = null;

    state.matched++;

    updateProgress();

    playSound(true);

    showToast(true, "啱喇！" + pair.tip);


    if (state.matched === pairs.length) {

      state.locked = true;

      setTimeout(showWin, 750);

    }

  } else {

    state.wrong++;

    state.wrongIds.add(idL);

    state.locked = true;


    l.classList.add("wrong");

    r.classList.add("wrong");

    playSound(false);


    showToast(false, `唔啱喎。「${pair.left}」應該配「${pair.right}」。${pair.tip}`);


    setTimeout(() => {

      l.classList.remove("wrong", "selected");

      r.classList.remove("wrong", "selected");

      state.selectedLeft = null;

      state.selectedRight = null;

      state.locked = false;

    }, 1000);

  }

}

function renderOrder(board) {

  board.className = "order-wrap";

  board.innerHTML = `

    <div class="order-target" id="orderTarget">

    <div class="order-pool" id="orderPool"></div>

  `;


  const pool = $("orderPool");

  const seq = state.currentSequence;

  const shuffled = shuffle(seq.map((s, i) => ({ ...s, id: i })));


  shuffled.forEach(item => {

    const div = document.createElement("div");

    div.className = "order-line";

    div.dataset.id = item.id;

        <div class="line-jyut">${item.jyut}</div>

      </div>

    `;

    div.addEventListener("click", () => onOrderClick(div));

    pool.appendChild(div);

  });

}


function onOrderClick(el) {

  if (state.locked) return;

  if (el.classList.contains("done")) return;


  const id = +el.dataset.id;

  const seq = state.currentSequence;


  if (id === state.orderIndex) {

    // 正確

    el.classList.add("done");

    el.style.cursor = "default";

    const target = $("orderTarget");

    const hint = target.querySelector(".empty-hint");

    if (hint) hint.remove();


    const item = seq[id];

    const row = document.createElement("div");

    row.className = "order-line done";

    row.innerHTML = `

      <span class="order-num">${state.orderIndex + 1}</span>

      <span class="role-tag">${item.role}</span>

      <div class="line-body">

        <div class="line-text">${item.text}</div>

        <div class="line-jyut">${item.jyut}</div>

      </div>

    `;

    target.appendChild(row);


    // 從 pool 移除

    el.remove();


    state.orderIndex++;

    updateProgress();

    playSound(true);

    showToast(true, `第 ${state.orderIndex} 句啱喇！${item.tip}`);


    if (state.orderIndex === seq.length) {

      state.locked = true;

      setTimeout(showWin, 750);

    }

  } else {

    // 錯誤

    state.wrong++;

    state.wrongIds.add(state.orderIndex);

    state.locked = true;

    el.classList.add("wrong");

    playSound(false);

    const correct = seq[state.orderIndex];

    showToast(false, `次序唔啱喎。下一個應該係「${correct.role}：${correct.text}」。`);


    setTimeout(() => {

      el.classList.remove("wrong");

      state.locked = false;

    }, 1000);

  }

}

function renderListenQuestion() {

  const board = $("board");

  if (!q) return;


  // 打亂選項

  const opts = shuffle(q.options.slice());


  board.innerHTML = `

    <div class="listen-hint">點喇叭聽粵語，再選出正確嘅粵拼<br><span style="font-size:0.85em;color:#b0a090;">（可重複點擊播放）</span></div>

    <div class="options" id="listenOptions"></div>

  `;


  const speaker = $("speakerBtn");

  speaker.addEventListener("click", () => speakCantonese(q.text, speaker));


  const optWrap = $("listenOptions");

  opts.forEach(o => {

    const b = document.createElement("button");

    b.className = "option-btn";

    b.textContent = o;

    b.dataset.value = o;

    optWrap.appendChild(b);

  });


  // 自動播放一次（若瀏覽器允許）

  setTimeout(() => {

    try { speakCantonese(q.text, speaker); } catch (e) {}

  }, 350);


  updateProgress();

}


function onListenAnswer(btn) {

  if (state.locked) return;

  const q = state.currentQuestions[state.qIndex];

  const val = btn.dataset.value;


  if (val === q.answer) {

    btn.classList.add("correct");

    state.locked = true;

    playSound(true);

    showToast(true, `啱喇！「${q.text}」讀 ${q.answer}。${q.tip}`);


    setTimeout(() => {

      state.qIndex++;

      state.locked = false;

      if (state.qIndex >= state.currentQuestions.length) {

        showWin();

      } else {

        renderListenQuestion();

      }

    }, 1300);

  } else {

    btn.classList.add("wrong");

    btn.disabled = true;

    state.wrong++;

    state.wrongIds.add(state.qIndex);

    playSound(false);

    showToast(false, `唔啱喎。正確答案係 ${q.answer}。${q.tip}`);

  }

}

function updateProgress() {

  const lv = LEVELS[state.levelIndex];

  let done = 0, total = 0;


  if (lv.type === "match") {

    done = state.matched;

    total = state.currentPairs.length;

  } else if (lv.type === "order") {

    done = state.orderIndex;

    total = state.currentSequence.length;

  } else if (lv.type === "listen") {

    done = state.qIndex;

    total = state.currentQuestions.length;

  }


  $("progressText").textContent = done + " / " + total;

  $("progressFill").style.width = total ? (done / total * 100) + "%" : "0%";

}

function showWin() {

  stopTimer();

  const lv = LEVELS[state.levelIndex];

  const wrong = state.wrong;

  const time = state.elapsed;


  const stars = wrong <= 1 ? 3 : wrong <= 3 ? 2 : 1;

  const isRetry = state.mode === "retry";


  $("winEmoji").textContent = stars === 3 ? "🏆" : stars === 2 ? "🎉" : "👍";

  $("winTitle").textContent = stars === 3 ? "完美通關！" : stars === 2 ? "恭喜闖關成功！" : "過關喇！";


  const starEls = [1, 2, 3].map(n =>

    `<span class="${n <= stars ? 'on' : 'dim'}" style="animation-delay:${n * 0.18}s">⭐</span>`

  ).join("");

  $("winStars").innerHTML = starEls;


  $("winMsg").textContent =

    stars === 3 ? "你全部一次過配對正確，好叻呀！" :

    stars === 2 ? "表現唔錯，再練習一次可以攞三粒星！" :

    "完成咗喇！記住錯過嘅詞語，再挑戰一次。";


  $("winDetail").innerHTML =

    `關卡：<strong>${lv.emoji} ${lv.name}</strong>${isRetry ? "（錯題重練）" : ""}<br>` +

    `用時：<strong>${fmtTime(time)}</strong>　錯誤：<strong>${wrong}</strong> 次<br>` +

    `<span style="color:#8a7a6a;font-size:14px;">${lv.subtitle}</span>`;


  // 名字輸入

  const nameRow = $("nameRow");

  if (isRetry) {

    nameRow.style.display = "none";

  } else {

    nameRow.style.display = "flex";

    const last = localStorage.getItem("yq_player_name") || "";

    $("nameInput").value = last;

    $("saveScoreBtn").onclick = () => {

      const name = ($("nameInput").value || "匿名同學").trim().slice(0, 12);

      localStorage.setItem("yq_player_name", name);

      addScore(state.levelIndex, {

        name, time, stars, wrong,

        date: new Date().toLocaleDateString("zh-HK")

      });

      $("saveScoreBtn").textContent = "已保存 ✓";

      $("saveScoreBtn").disabled = true;

      $("saveScoreBtn").style.opacity = "0.7";

    };

    $("saveScoreBtn").textContent = "保存成績";

    $("saveScoreBtn").disabled = false;

    $("saveScoreBtn").style.opacity = "1";

  }


  // 下一關

  const nextBtn = $("nextBtn");

  if (state.levelIndex >= LEVELS.length - 1) {

    nextBtn.textContent = "返回地圖";

    nextBtn.onclick = () => { closeWin(); backHome(); };

  } else {

    nextBtn.textContent = "下一關 →";

    nextBtn.onclick = () => {

      closeWin();

      startLevel(state.levelIndex + 1, "normal");

    };

  }


  // 重練錯題

  const retryBtn = $("retryBtn");

  const hasWrong = state.wrongIds.size > 0;

  if (hasWrong && lv.type === "match" && !isRetry) {

    retryBtn.style.display = "inline-block";

    retryBtn.textContent = `重練錯題（${state.wrongIds.size}）`;

    retryBtn.onclick = () => {

      closeWin();

      startLevel(state.levelIndex, "retry", Array.from(state.wrongIds));

    };

  } else if (hasWrong && lv.type !== "match" && !isRetry) {

    retryBtn.style.display = "inline-block";

    retryBtn.textContent = "再練一次";

    retryBtn.onclick = () => {

      closeWin();

      startLevel(state.levelIndex, "retry");

    };

  } else {

    retryBtn.style.display = "none";

  }


  $("winModal").classList.add("show");

  confetti();

  playSound(true);

}

function escapeHtml(s) {

  return String(s).replace(/[&<>"']/g, c => ({

    "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;"

  }[c]));

}
